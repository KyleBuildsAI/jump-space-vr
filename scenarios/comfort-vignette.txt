# Comfort vignette in the hangar: the view's edges darken while the stick moves or turns the player, not when standing
# still or moving only the head, and never when it is off.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
config WeaponInHand true
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
config ComfortVignette High
config TurnMode Smooth
wait 2
expect vignette < 0.02

# Walking with the stick darkens the edges; stopping clears them. (The spawn point faces a wall about a second's
# walk away, so check while still walking.)
input left.stick 0 1
wait 0.8
expect vignette > 0.2
snapshot
input left.stick 0 0
wait 2
expect vignette < 0.02

# Moving only the head (real movement) never does.
head 0.3 1.5 0.3  10 30 0
wait 0.2
head 0 1.6 0  0 0 0
wait 0.5
expect vignette < 0.02

# Smooth turning does.
input right.stick 1 0
wait 1
expect vignette > 0.2
input right.stick 0 0
wait 2
expect vignette < 0.02

# Off: nothing, even while walking.
config ComfortVignette Off
input left.stick 0 -1
wait 0.8
expect vignette < 0.02
input left.stick 0 0
