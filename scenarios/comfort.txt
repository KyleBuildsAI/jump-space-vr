# Comfort checks in the prologue level. Starts from the main menu and enters the prologue with the keyboard.

waitfor vrRendering == 1 120
head 0 1.6 0  0 0 0
hand right 0.2 1.3 0.3  0 0 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
key 0x28
key 0x26
key 0x0D
waitfor inLevel == 1 150
# The prologue opens with a wake-up cinematic; look input is locked until it ends.
waitfor cutscene == 1 90
waitfor cutscene == 0 300
wait 5
recenter
wait 2

# The character stands up after the cinematic; let that finish.
wait 8

# Comfort: walking with "Stabilize view" on keeps vertical jolts (bob, steps) out of the view. eyeJolt is the largest
# change in vertical eye speed between frames (cm/s). Measured on the prologue stairs: about 7 with it on, 230-335 off.
config StabilizeView true
input left.stick 0 1
wait 2
expect eyeJolt < 30
input left.stick 0 0
wait 2
config StabilizeView false
input left.stick 0 -1
wait 2
log for comparison, without stabilization:
expect eyeJolt > 0
input left.stick 0 0
