# First-person override in the hangar: in normal first-person play the view is never moved to the head point.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
wait 2

# First person: the camera is in the head and nothing is overridden.
expect firstPersonOverride == 0
expect cameraHeadDistanceCm < 60
# Standing in the hangar, "level" is the world.
expect frameTiltDeg < 1

# Walking around in first person never triggers it (in the prologue the game reports its third-person body as shown
# while the camera is first person; that once moved the view out of the head).
input left.stick 0 1
wait 3
input left.stick 0 0
wait 1
expect firstPersonOverride == 0
expect cameraHeadDistanceCm < 60
