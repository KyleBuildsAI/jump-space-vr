# In the hangar with a real gamepad (a scripted test pad): using it switches from motion controls to the pad, the
# view follows the game camera, firing rumbles the pad, View held opens the VR menu, and the VR controllers take
# over again as soon as they are used.
waitfor vrRendering == 1 120
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
config ControllerMode Auto
config WeaponInHand false
config Haptics true
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
wait 2

# Motion controls: aiming 30 degrees right turns the game camera but not the view.
expect gamepadActive == 0
expect motionActive == 1
hand right 0.18 1.3 0.35  0 30 0
wait 2
expect viewYawOffset == 30 4

# Touching the pad switches to it: no motion aim, the view follows the game camera again.
pad rightstick 0.6 0
wait 0.3
pad rightstick 0 0
wait 1
expect gamepadActive == 1
expect motionActive == 0
expect viewYawOffset == 0 2

# Firing with the pad rumbles it (the game sends no rumble of its own).
pad righttrigger 1
wait 0.45
pad righttrigger 0
expect padRumbles > 2

# The pad's left trigger aims down sights: the weapon shifts toward the aiming eye.
pad lefttrigger 1
wait 0.6
expect sightShiftCm > 1
pad lefttrigger 0
wait 0.5

# Room-scale walking, then running with the pad: the view must stay with the body, not fall behind it.
config RoomscaleWalking true
input right.trigger 1
wait 0.2
input right.trigger 0
head 0 1.6 0.35  20 0 0
wait 1
pad leftstick 0 1
wait 2.5
pad leftstick 0 0
wait 0.5
expect gamepadActive == 1
expect headLocalCm < 40
head 0 1.6 0  20 0 0
config RoomscaleWalking false
recenter
wait 1

# Holding View for a second opens the VR menu; the stick moves through it. Holding it again closes it.
pad view 1
wait 1.3
pad view 0
wait 0.3
expect modMenuOpen == 1
# The game's own input is held back while the menu is open (the pad's buttons would otherwise also jump, crouch...).
expect gameActionsEnabled == 0
# A/Cross opens the first category, B/Circle goes back to the categories.
pad a 1
wait 0.2
pad a 0
wait 0.3
expect modMenuOnPage == 1
pad b 1
wait 0.2
pad b 0
wait 0.3
expect modMenuOnPage == 0
expect modMenuOpen == 1
pad view 1
wait 1.3
pad view 0
wait 0.3
expect modMenuOpen == 0
expect gameActionsEnabled > 0

# Using the VR controllers again hands control back to them.
input right.trigger 1
wait 0.2
input right.trigger 0
wait 0.5
expect gamepadActive == 0
expect motionActive == 1

# Controls = Gamepad keeps the pad in charge whatever the controllers do.
config ControllerMode Gamepad
input right.trigger 1
wait 0.2
input right.trigger 0
wait 0.5
expect gamepadActive == 1
pad release
