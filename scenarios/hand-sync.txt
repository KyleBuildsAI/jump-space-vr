# Multiplayer hands in the hangar, without a second player: the loopback option feeds our own shared hands back
# through the wire format onto our own third-person body, where a remote player's hands would be shown.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
config WeaponInHand true
config Haptics true
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
config WeaponInHand true
config HandSyncLoopback true
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
hand left -0.25 1.0 0.1  0 0 0
wait 2

# Multiplayer hands, looped back: our shared hands go through the wire format and bend our own third-person arms the
# way a remote player's would be bent.
expect remoteHands == 1
expect remoteHandErrorCm < 6
snapshot

# Moving the hands moves the third-person arms with them.
hand right 0.3 1.5 0.3  0 30 0
hand left -0.3 1.3 0.3  0 0 0
wait 1
expect remoteHands == 1
expect remoteHandErrorCm < 6

# Without weapon in hand the controller positions are shared instead.
config WeaponInHand false
wait 1
expect remoteHands == 1
expect remoteHandErrorCm < 6

# Switched off, nothing is shown.
config ShareHands false
wait 1
expect remoteHands == 0
