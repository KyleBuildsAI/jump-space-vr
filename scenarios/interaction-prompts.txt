# Interaction prompts follow the head (look to interact), in the hangar with a gun. In both halves of each check
# the head looks at the object and the gun points away to the side: with look to interact off the game aims with
# the gun and selects nothing (or something else), with it on and the gun lowered the game selects what the
# player looks at. Covers the Discord feedback about grapple prompts and the galaxy map being hard to select.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
config WeaponInHand true
config BodyFollowsHead false
recenter
head 0 1.6 0  0 0 0
hand right 0.3 1.15 0.1  65 70 0
hand left -0.25 1.0 0.1  0 0 0
wait 2

# Galaxy map table, checked first so the player comes to it from the hangar spawn (from the far side the game
# offers no interaction). With the gun pointing away the old behaviour picks whatever the gun points at instead.
target GalaxyMap
goto 1.8
wait 3
face
wait 1
recenter
wait 1
config LookToInteract false
hand right 0.3 1.3 0.2  0 70 0
wait 1
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 1.5
print onTarget hasInteractTarget targetDistance targetPitch gazeAim cameraPitch
expect onTarget == 0
config LookToInteract true
hand right 0.3 1.15 0.1  65 70 0
wait 1
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 1.5
print onTarget hasInteractTarget targetDistance targetPitch gazeAim cameraPitch
expect onTarget == 1

# Grapple point: its range is 10-30 m and it only accepts players about level with it or above, with a clear view.
target GrapplingPoint PF_GrapplePoint_01
goto 15 spot
wait 2
face
recenter
wait 0.5
targetinfo

config LookToInteract false
hand right 0.3 1.3 0.2  0 70 0
wait 1
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 1.5
print grappleAimed targetDistance targetPitch gazeAim cameraPitch
expect grappleAimed == 0
config LookToInteract true
hand right 0.3 1.15 0.1  65 70 0
wait 1
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 2
headlook targetYaw targetPitch
wait 1.5
print grappleAimed targetDistance targetPitch gazeAim cameraPitch
expect grappleAimed == 1
