# Look to interact and body follows head, in the hangar with a gun: lowering the gun makes the game look where the
# player looks (prompts and interaction targets follow the gaze), raising it or firing aims with the gun again, and
# turning the head well away from the body turns the body after it without moving the view.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
config WeaponInHand true
config LookToInteract true
config BodyFollowsHead true
recenter
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
wait 2

# Gun raised: the game looks along the gun.
expect gazeAim == 0
hand right 0.18 1.3 0.35  0 -20 0
wait 1
expect aimYawOffset == -20 3

# Gun lowered (pointing down): the game looks where the head looks.
hand right 0.18 1.2 0.2  60 -20 0
head 0 1.6 0  -10 20 0
wait 1
expect gazeAim == 1
expect aimYawOffset == 20 3
expect cameraPitch == 10 3
snapshot

# Pulling the trigger hands the look straight back to the gun.
input right.trigger 1
wait 0.2
expect gazeAim == 0
input right.trigger 0
wait 0.5
expect gazeAim == 1

# Raising the gun again aims with it.
hand right 0.18 1.3 0.35  0 -20 0
wait 1
expect gazeAim == 0
expect aimYawOffset == -20 3

# Body follows head: a small head turn leaves the body alone...
head 0 1.6 0  0 20 0
wait 1
expect bodyFollowing == 0
expect headYawLocal == 20 2
# ...turning the head 70 degrees turns the body after it until it faces the head again; the view stays put.
print eyeYaw bodyYaw
head 0 1.6 0  0 70 0
wait 2
expect bodyFollowing == 0
expect headYawLocal == 0 5
print eyeYaw bodyYaw

# Off: the body stays.
config BodyFollowsHead false
head 0 1.6 0  0 0 0
wait 1
recenter
wait 1
head 0 1.6 0  0 70 0
wait 2
expect headYawLocal == 70 3
