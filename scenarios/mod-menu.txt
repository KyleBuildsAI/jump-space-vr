# The VR settings menu: opens with the controller gesture (hold right stick click), navigates with the left stick,
# toggles with A, closes with B. Runs on the main menu (about 1 minute).

waitfor vrRendering == 1 120
head 0 1.6 0  0 0 0
hand right 0.2 1.3 -0.3  0 0 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 2

input right.stickclick 1
wait 1.4
input right.stickclick 0
wait 0.5
expect modMenuOpen == 1
