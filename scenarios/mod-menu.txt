# The VR settings menu: opens with the controller gesture (hold right stick click) on the main page of categories,
# A opens a category, the left stick moves through it, A toggles, B goes back and then closes. Runs on the main menu
# (about 1 minute).

waitfor vrRendering == 1 120
head 0 1.6 0  0 0 0
hand right 0.2 1.3 -0.3  0 0 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 2

input right.stickclick 1
wait 1.4
input right.stickclick 0
wait 0.5
expect modMenuOpen == 1
expect modMenuOnPage == 0

# "View and comfort" is the first category; opening it selects its first setting, "Level horizon".
input right.a 1
wait 0.2
input right.a 0
wait 0.3
expect modMenuOnPage == 1
expect setting.LevelHorizon == 1
input right.a 1
wait 0.2
input right.a 0
wait 0.3
expect setting.LevelHorizon == 0
input right.a 1
wait 0.2
input right.a 0
wait 0.3
expect setting.LevelHorizon == 1

# Down one: "Remove head bob and shake" toggles with left/right on the stick (and ends unchanged).
input left.stick 0 -1
wait 0.15
input left.stick 0 0
wait 0.25
expect setting.StabilizeView == 1
input left.stick 1 0
wait 0.15
input left.stick 0 0
wait 0.25
expect setting.StabilizeView == 0
input left.stick 1 0
wait 0.15
input left.stick 0 0
wait 0.25
expect setting.StabilizeView == 1

# B goes back to the categories, B again closes the menu.
input right.b 1
wait 0.2
input right.b 0
wait 0.3
expect modMenuOpen == 1
expect modMenuOnPage == 0
input right.b 1
wait 0.2
input right.b 0
wait 0.3
expect modMenuOpen == 0
