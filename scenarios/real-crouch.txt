# Real crouch in the hangar: ducking 45 cm makes the character crouch and the view stays at the real head height (the
# game's own crouch camera drop is undone); standing up stands it up; switched off, ducking does nothing in the game.
waitfor vrRendering == 1 120
# Default UI panel placement, so the laser lands on the same menu item whatever the player has set.
config UiFollow Lazy
config UiDistance 1.8
config UiWidth 2.4
head 0 1.6 0  0 0 0
hand right 0 1.6 0  7.1 -25 0
recenter
key 0x0D
wait 8
keyuntil 0x1B 4 mainMenu == 1 90
wait 3
# "Enter Hangar", the second menu item.
expect laserV == 0.39 0.012
input right.trigger 1
wait 0.3
input right.trigger 0
# The game asks to confirm entering the hangar before the prologue has been played; F confirms.
wait 3
key 0x46
waitfor holdsItem == 1 360
wait 20
config WeaponInHand true
config RealCrouch true
head 0 1.6 0  0 0 0
hand right 0.18 1.3 0.35  0 0 0
recenter
wait 3
expect gameCrouching == 0
print eyeY

# Duck.
head 0 1.15 0  0 0 0
wait 2
expect realCrouch == 1
expect gameCrouching == 1
expect crouchLiftCm > 15
print eyeY crouchLiftCm
snapshot

# Stand up.
head 0 1.6 0  0 0 0
wait 2
expect realCrouch == 0
expect gameCrouching == 0
print eyeY

# Off: ducking only lowers the view.
config RealCrouch false
head 0 1.15 0  0 0 0
wait 2
expect gameCrouching == 0
head 0 1.6 0  0 0 0
